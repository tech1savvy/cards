# AWS Database Types

- ==1;;Amazon RDS / Aurora==
	- type: ==2;;Relational==
	- uses: ERP, CRM, financial apps

- ==1;;Amazon DynamoDB==
	- type: ==2;;Key-Value==
	- uses: real-time bidding, gaming, IoT

- ==1;;Amazon DocumentDB==
	- type: ==2;;Document==
	- uses: content management, user profiles

- ==1;;Amazon Neptune==
	- type: ==2;;Graph==
	- uses: social networks, fraud detection, recommendations

- ==1;;Amazon QLDB==
	- type: ==2;;Ledger==
	- uses: banking, supply chain, immutable records

- ==1;;Amazon ElastiCache / MemoryDB==
	- type: ==2;;In-Memory DB==
	- uses: caching, real-time leaderboards, session stores

- ==1;;Amazon Timestream==
	- type: ==2;;Time-Series==
	- uses: IoT telemetry, DevOps monitoring, industrial data

- ==1;;Amazon Keyspaces==
	- type: ==2;;Wide Column==
	- uses: high-volume time-series, device data, industrial apps

- ==1;;Amazon OpenSearch Service==
	- type: ==2;;Search==
	- uses: full-text search, log analytics, observability
